    75→  const [pdfBusy, setPdfBusy] = useState(false);

    78→  const [isolatedGroup, setIsolatedGroup] = useState<string | null>(null);
    79→  const [focusRequest, setFocusRequest] = useState<{ id: string; nonce: number } | null>(null);
    80→  const [snapshotSignal, setSnapshotSignal] = useState(0);
    81→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
    82→  const [showDimensions, setShowDimensions] = useState(true);
    83→  const [measureMode, setMeasureMode] = useState(false);
    84→  const [measureResult, setMeasureResult] = useState<{ distance: number } | null>(null);
    85→  const [comparingRev, setComparingRev] = useState<number | null>(null);

    91→
    92→  // arraste da alça de redimensionamento do dock (desktop)
    93→  useEffect(() => {
    94→    const onMove = (e: MouseEvent) => {
    95→      const drag = dockDragRef.current;
    96→      if (!drag) return;
    97→      const maxW = Math.min(DOCK_MAX, Math.max(DOCK_MIN, Math.floor(window.innerWidth * 0.6)));
    98→      const w = Math.min(maxW, Math.max(DOCK_MIN, drag.startW + (drag.startX - e.clientX)));
    99→      dockWidthRef.current = w; // sync imediato — efeito passivo pode atrasar em headless
   100→      setDockWidth(w);
   101→    };
   102→    const onUp = () => {
   103→      if (!dockDragRef.current) return;
   104→      dockDragRef.current = null;
   105→      document.body.style.cursor = "";
   106→      document.body.style.userSelect = "";
   107→      try { localStorage.setItem(DOCK_WIDTH_KEY, String(dockWidthRef.current)); } catch { /* ignore */ }
   108→    };
   109→    window.addEventListener("mousemove", onMove);
   110→    window.addEventListener("mouseup", onUp);
   111→    return () => {
   112→      window.removeEventListener("mousemove", onMove);
   113→      window.removeEventListener("mouseup", onUp);
   114→    };
   115→  }, []);

   119→      ...h.slice(-59),
   120→      { ...item, id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`, ts: new Date().toISOString() },
   121→    ]);
   122→  }, []);
   123→
   124→  // restaura histórico do localStorage
   125→  useEffect(() => {
   126→    try {
   127→      const raw = localStorage.getItem(HISTORY_KEY);
   128→      if (raw) setHistory(JSON.parse(raw) as HistoryItem[]);
   129→      const w = localStorage.getItem(DOCK_WIDTH_KEY);
   130→      if (w) {
   131→        const n = Number.parseInt(w, 10);
   132→        if (Number.isFinite(n) && n >= DOCK_MIN && n <= DOCK_MAX) setDockWidth(n);
   133→      }
   134→    } catch {
   135→      // ignora storage corrompido
   136→    }
   137→  }, []);
   138→
   139→  const refresh = useCallback(async () => {
   140→    try {
   141→      const st = await api.getProject();
   142→      setProjectState(st);
   143→    } catch (e) {
   144→      toast.error((e as ApiCallError).message);
   145→    }
   146→  }, []);
   147→
   148→  useEffect(() => {
   149→    void refresh();

   151→      .meta()
   152→      .then((m) => {
   153→        const active = m.providers.find((p) => p.active) ?? m.providers[0];
   154→        setProviderLabel(`${active?.label ?? "mock"} · ${active?.current_model ?? ""}`);
   155→        setActiveVision(active?.supports_image ?? true);
   156→      })
   157→      .catch(() => setProviderLabel("mock · mock-transformer-v1"));
   158→  }, [refresh]);
   159→
   160→  // Ctrl+Z → undo atômico; 1–7 → vistas; Esc → desselecionar
   161→  useEffect(() => {
   162→    const onKey = (e: KeyboardEvent) => {
   163→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "z") {
   164→        e.preventDefault();
   165→        void doUndo();
   166→        return;
   167→      }
   168→      if (e.key === "Escape") {
   169→        setSelectedId(null);
   170→        setMeasureMode(false);
   171→        setMeasureResult(null);
   172→        return;
   173→      }
   174→      const target = e.target as HTMLElement | null;
   175→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable)) return;
   176→      const viewKeys = ["1", "2", "3", "4", "5", "6", "7"];
   177→      const idx = viewKeys.indexOf(e.key);
   178→      if (idx >= 0) setViewMode(VIEWS[idx]);
   179→      if (e.key.toLowerCase() === "d") setShowDimensions((v) => !v);
   180→      if (e.key.toLowerCase() === "m") setMeasureMode((v) => !v);
   181→    };
   182→    window.addEventListener("keydown", onKey);
   183→    return () => window.removeEventListener("keydown", onKey);
   184→  }, []);
   185→

   188→      const st = await api.undo();

   190→      setCandidate(null);
   191→      setRestoreInfo(null);
   192→      setHistoryRefreshKey((k) => k + 1);
   193→      pushHistory({ role: "info", text: st.message ?? "desfeito" });
   194→      toast.success("Desfeito — JSON anterior restaurado");

   196→      const err = e as ApiCallError;
   197→      if (err.payload?.type === "nothing_to_undo") toast.info("Nada para desfazer");
   198→      else toast.error(err.payload?.message ?? err.message);
   199→    }
   200→  }, [pushHistory]);
   201→
   202→  // ---------- transform via IA ----------
   203→  const sendRequest = useCallback(
   204→    async (text: string, attachments: Array<{ type: "image"; name: string; data_url: string }>) => {
   205→      if (!projectState) return;
   206→      pushHistory({ role: "user", text });

   208→      setCandidate(null);
   209→      try {
   210→        const r: TransformResult = await api.transform({
   211→          request: text,
   212→          base_revision: projectState.revision,
   213→          base_hash: projectState.hash,

   215→        });

   217→          setCandidate({
   218→            project: r.candidate_project,
   219→            diff: (r.diff ?? { added: [], removed: [], modified: [], counts: { added: 0, removed: 0, modified: 0, total: 0 }, panel_changed: false, info_changed: false }),
   220→            diffSummary: r.diff_summary ?? "",
   221→            baseRevision: r.base_revision,
   222→            baseHash: r.base_hash,
   223→            hash: r.candidate_hash,
   224→          });
   225→          pushHistory({
   226→            role: "assistant",
   227→            text: r.explain,
   228→            meta: {
   229→              status: "ready",
   230→              explain: r.explain,
   231→              assumptions: r.assumptions,
   232→              questions: r.questions,
   233→              diffSummary: r.diff_summary,
   234→              provider: r.meta.provider,
   235→              model: r.meta.model,
   236→              latency: r.meta.latency_ms,
   237→              fewShot: r.meta.few_shot_id ?? null,
   238→              candidate: r.candidate_project,
   239→            },
   240→          });
   241→        } else {
   242→          pushHistory({
   243→            role: "assistant",
   244→            text: r.explain,
   245→            meta: {
   246→              status: "needs_input",
   247→              explain: r.explain,
   248→              questions: r.questions,
   249→              assumptions: r.assumptions,
   250→              provider: r.meta.provider,
   251→              model: r.meta.model,
   252→              latency: r.meta.latency_ms,
   253→              fewShot: r.meta.few_shot_id ?? null,
   254→            },
   255→          });
   256→        }
   257→      } catch (e) {
   258→        const err = e as ApiCallError;
   259→        const payload = err.payload;
   260→        pushHistory({
   261→          role: "error",
   262→          text: ERRORS_FRIENDLY[payload?.type ?? ""] ?? payload?.message ?? err.message,
   263→          meta: { errorType: payload?.type ?? "error" },
   264→        });
   265→      } finally {
   266→        setSending(false);
   267→      }
   268→    },
   269→    [projectState, pushHistory],
   270→  );
   271→

   276→        const r = await api.preview(doc);
   277→        if (r.ok && r.diff && projectState) {
   278→          setCandidate({
   279→            project: doc,
   280→            diff: r.diff,
   281→            diffSummary: r.diff_summary ?? "",
   282→            baseRevision: r.base_revision ?? projectState.revision,
   283→            baseHash: r.base_hash ?? projectState.hash,
   284→            hash: r.candidate_hash ?? "",
   285→          });
   286→          pushHistory({ role: "info", text: `JSON validado — ${r.diff_summary}. Revise o preview 3D e aplique.` });
   287→        }
   288→        return r;
   289→      } catch (e) {
   290→        const err = e as ApiCallError;
   291→        toast.error(err.payload?.message ?? err.message);
   292→        return { ok: false, errors: [{ path: "preview", message: err.message }] };
   293→      }
   294→    },
   295→    [projectState, pushHistory],
   296→  );
   297→
   298→  const applyCandidate = useCallback(async () => {
   299→    if (!candidate) return;
   300→    try {
   301→      // restauração de revisão usa a rota dedicada (source "restore" no log)
   302→      const st = restoreInfo !== null
   303→        ? await api.restoreRevision(restoreInfo)
   304→        : await api.apply({
   305→            candidate: candidate.project,
   306→            base_revision: candidate.baseRevision,
   307→            base_hash: candidate.baseHash,
   308→          });

   310→      setCandidate(null);
   311→      setRestoreInfo(null);
   312→      setHistoryRefreshKey((k) => k + 1);
   313→      setHistory((h) =>
   314→        h.map((item) =>
   315→          item.role === "assistant" && item.meta?.candidate && !item.meta.applied
   316→            ? { ...item, meta: { ...item.meta, applied: true } }
   317→            : item,
   318→        ),
   319→      );
   320→      pushHistory({
   321→        role: "info",
   322→        text: restoreInfo !== null
   323→          ? `Revisão ${restoreInfo} restaurada como revision ${st.revision}.`
   324→          : `Candidato aplicado — revision ${st.revision}.`,

   326→      toast.success(restoreInfo !== null ? `Revisão ${restoreInfo} restaurada — rev ${st.revision}` : `Aplicado — revision ${st.revision}`);
   327→    } catch (e) {
   328→      const err = e as ApiCallError;
   329→      if (err.status === 409) {
   330→        toast.error("O projeto mudou desde o preview — regenere o candidato.");

   332→        setRestoreInfo(null);
   333→        void refresh();
   334→      } else {
   335→        toast.error(err.payload?.message ?? err.message);
   336→      }
   337→    }
   338→  }, [candidate, restoreInfo, pushHistory, refresh]);
   339→
   340→  // ---------- comparação com revisão do histórico ----------

   352→          diffSummary: r.diff_summary ?? "",
   353→          baseRevision: r.base_revision,
   354→          baseHash: r.base_hash,
   355→          hash: r.candidate_hash ?? "",

   359→          role: "info",
   360→          text: `Comparando rev ${rev} com a atual — ${r.diff_summary}. APLICAR restaura a rev ${rev}; CANCELAR descarta.`,

   366→      toast.error((e as ApiCallError).payload?.message ?? (e as ApiCallError).message);

   368→      setComparingRev(null);
   369→    }

   372→  const cancelCandidate = useCallback(() => {

   377→
   378→  // ---------- status maps para o viewer ----------

   381→    if (!candidate) return { statuses: {}, candidateStatuses: {} };

   386→    const removed = new Set(candidate.diff.removed);

   390→    }
   391→    const sNew: StatusMap = {};
   392→    for (const id of newIds) {
   393→      sNew[id] = added.has(id) ? "added" : modified.has(id) ? "modified" : "unchanged";
   394→    }
   395→    return { statuses: sOld, candidateStatuses: sNew };
   396→  }, [projectState, candidate]);
   397→
   398→  const selectedElement = useMemo(() => {
   399→    const doc = candidate?.project ?? projectState?.project;
   400→    return doc?.elements.find((e) => e.id === selectedId) ?? null;
   401→  }, [projectState, candidate, selectedId]);
   402→
   403→  const weightInfo = useMemo(
   404→    () => (selectedElement ? estimateElementWeightKg(selectedElement) : null),
   405→    [selectedElement],
   406→  );
   407→
   408→  const actions = {
   409→    newProject: async () => {
   410→      try {
   411→        const st = await api.newProject();
   412→        setProjectState(st);
   413→        setCandidate(null);
   414→        setRestoreInfo(null);
   415→        setHistoryRefreshKey((k) => k + 1);
   416→        pushHistory({ role: "info", text: "Novo projeto em branco criado." });
   417→      } catch (e) {
   418→        toast.error((e as ApiCallError).message);
   419→      }
   420→    },
   421→    importJson: async (file: File) => {
   422→      try {
   423→        const text = await file.text();
   424→        const doc = JSON.parse(text);
   425→        const st = await api.importProject(doc);
   426→        setProjectState(st);
   427→        setCandidate(null);
   428→        setRestoreInfo(null);
   429→        setHistoryRefreshKey((k) => k + 1);
   430→        pushHistory({ role: "info", text: "JSON importado e validado." });
   431→        toast.success("Projeto importado");
   432→      } catch (e) {
   433→        toast.error((e as ApiCallError).payload?.message ?? (e as Error).message);
   434→      }
   435→    },
   436→    exportPdf: async () => {

   438→      try {
   439→        await api.downloadPdf();
   440→        toast.success("PDF gerado");
   441→      } catch (e) {
   442→        toast.error((e as ApiCallError).payload?.message ?? (e as Error).message);
   443→      } finally {
   444→        setPdfBusy(false);
   445→      }
   446→    },
   447→  };
   448→
   449→  // persiste histórico (sem candidatos antigos pesados)
   450→  useEffect(() => {

   452→      const slim = history.slice(-30).map((h) => ({
   453→        ...h,
   454→        meta: h.meta?.candidate ? { ...h.meta, candidate: null } : h.meta,
   455→      }));
   456→      localStorage.setItem(HISTORY_KEY, JSON.stringify(slim));
   457→    } catch {
   458→      // quota — ignora
   459→    }
   460→  }, [history]);
   461→
   462→  const toggleHidden = useCallback((id: string) => {
   463→    setHiddenIds((s) => {
   464→      const n = new Set(s);
   465→      if (n.has(id)) n.delete(id);
   466→      else n.add(id);
   467→      return n;
   468→    });
   469→  }, []);
   470→
   471→  const isolateGroup = useCallback((group: string | null) => {
   472→    setIsolatedGroup(group);
   473→    if (!projectState) return;
   474→    const doc = candidate?.project ?? projectState.project;
   475→    if (group === null) {
   476→      setHiddenIds(new Set());
   477→    } else {
   478→      setHiddenIds(new Set(doc.elements.filter((e) => e.group !== group).map((e) => e.id)));
   479→    }
   480→  }, [projectState, candidate]);
   481→
   482→  const focusElement = useCallback((id: string) => {
   483→    setSelectedId(id);
   484→    setFocusRequest({ id, nonce: Date.now() });
   485→  }, []);
   486→
   487→  const fitKey = `${projectState?.revision ?? 0}:${projectState?.hash?.slice(0, 8) ?? ""}:${candidate?.hash.slice(0, 8) ?? "none"}`;
   488→  const project = projectState?.project;
   489→
   490→  return (
   491→    <TooltipProvider>
   492→      <div className="h-screen flex flex-col bg-slate-100 text-slate-900">
   493→        {/* ---------- TOPBAR ---------- */}
   494→        <header className="flex items-center gap-2 px-3 sm:px-4 h-14 bg-gradient-to-r from-[#141f2b] via-[#1b2836] to-[#223344] text-slate-100 shrink-0 border-b-4 border-orange-600 shadow-md" role="banner">
   495→          <div className="flex items-center gap-2.5 min-w-0">
   496→            <div className="h-8 w-8 rounded bg-orange-600 grid place-items-center font-black text-white text-sm shrink-0" aria-hidden>
   497→              LC
   498→            </div>
   499→            <div className="leading-tight min-w-0">
   500→              <div className="font-bold text-sm tracking-wide truncate">LED JSON CAD</div>
   501→              <div className="text-[10px] text-slate-400 truncate">documento JSON · mm · Z solo</div>
   502→            </div>
   503→          </div>
   504→

  1005→  const [open, setOpen] = useState(false);
