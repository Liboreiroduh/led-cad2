    64→  const [candidate, setCandidate] = useState<CandidateState | null>(null);
    65→  const [history, setHistory] = useState<HistoryItem[]>([]);
    66→  const [sending, setSending] = useState(false);
    67→  const [selectedId, setSelectedId] = useState<string | null>(null);
    68→  const [viewMode, setViewMode] = useState<ViewMode>("iso");
    69→  const [tab, setTab] = useState<DockTab>("copiloto");
    70→  const [providerModalOpen, setProviderModalOpen] = useState(false);
    71→  const [bomOpen, setBomOpen] = useState(false);
    72→  const [providerLabel, setProviderLabel] = useState("carregando…");
    73→  const [pdfBusy, setPdfBusy] = useState(false);
    74→  const importFileRef = useRef<HTMLInputElement>(null);
    75→  const [hiddenIds, setHiddenIds] = useState<Set<string>>(new Set());
    76→  const [isolatedGroup, setIsolatedGroup] = useState<string | null>(null);
    77→  const [focusRequest, setFocusRequest] = useState<{ id: string; nonce: number } | null>(null);
    78→  const [snapshotSignal, setSnapshotSignal] = useState(0);
    79→  const [shortcutsOpen, setShortcutsOpen] = useState(false);
    80→  const [showDimensions, setShowDimensions] = useState(true);
    81→  const [measureMode, setMeasureMode] = useState(false);
    82→  const [measureResult, setMeasureResult] = useState<{ distance: number } | null>(null);
    83→  const [comparingRev, setComparingRev] = useState<number | null>(null);
    84→  const [restoreInfo, setRestoreInfo] = useState<number | null>(null);
    85→  const [historyRefreshKey, setHistoryRefreshKey] = useState(0);
    86→  const [dockWidth, setDockWidth] = useState(DOCK_DEFAULT);
    87→  const dockDragRef = useRef<{ startX: number; startW: number } | null>(null);
    88→
    89→  const pushHistory = useCallback((item: Omit<HistoryItem, "id" | "ts">) => {
    90→    setHistory((h) => [
    91→      ...h.slice(-59),
    92→      { ...item, id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`, ts: new Date().toISOString() },
    93→    ]);
    94→  }, []);
    95→
    96→  // restaura histórico do localStorage
    97→  useEffect(() => {
    98→    try {
    99→      const raw = localStorage.getItem(HISTORY_KEY);
   100→      if (raw) setHistory(JSON.parse(raw) as HistoryItem[]);
   101→      const w = localStorage.getItem(DOCK_WIDTH_KEY);
   102→      if (w) {
   103→        const n = Number.parseInt(w, 10);
   104→        if (Number.isFinite(n) && n >= DOCK_MIN && n <= DOCK_MAX) setDockWidth(n);
   105→      }
   106→    } catch {
   107→      // ignora storage corrompido
   108→    }
   109→  }, []);
   110→
   111→  const refresh = useCallback(async () => {
   112→    try {
   113→      const st = await api.getProject();
   114→      setProjectState(st);
   115→    } catch (e) {
   116→      toast.error((e as ApiCallError).message);
   117→    }
   118→  }, []);
   119→
   120→  useEffect(() => {
   121→    void refresh();
   122→    void api
   123→      .meta()
   124→      .then((m) => {
   125→        const active = m.providers.find((p) => p.active) ?? m.providers[0];
   126→        setProviderLabel(`${active?.label ?? "mock"} · ${active?.current_model ?? ""}`);
   127→      })
   128→      .catch(() => setProviderLabel("mock · mock-transformer-v1"));
   129→  }, [refresh]);
   130→
   131→  // Ctrl+Z → undo atômico; 1–7 → vistas; Esc → desselecionar
   132→  useEffect(() => {
   133→    const onKey = (e: KeyboardEvent) => {
   134→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "z") {
   135→        e.preventDefault();
   136→        void doUndo();
   137→        return;
   138→      }
   139→      if (e.key === "Escape") {
   140→        setSelectedId(null);
   141→        setMeasureMode(false);
   142→        setMeasureResult(null);
   143→        return;
   144→      }
   145→      const target = e.target as HTMLElement | null;
   146→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable)) return;
   147→      const viewKeys = ["1", "2", "3", "4", "5", "6", "7"];
   148→      const idx = viewKeys.indexOf(e.key);
   149→      if (idx >= 0) setViewMode(VIEWS[idx]);
   150→      if (e.key.toLowerCase() === "d") setShowDimensions((v) => !v);
   151→      if (e.key.toLowerCase() === "m") setMeasureMode((v) => !v);
   152→    };
   153→    window.addEventListener("keydown", onKey);
   154→    return () => window.removeEventListener("keydown", onKey);
   155→  }, []);
   156→
   157→  const doUndo = useCallback(async () => {
   158→    try {
   159→      const st = await api.undo();
   160→      setProjectState(st);
   161→      setCandidate(null);
   162→      setRestoreInfo(null);
   163→      setHistoryRefreshKey((k) => k + 1);
   164→      pushHistory({ role: "info", text: st.message ?? "desfeito" });
   165→      toast.success("Desfeito — JSON anterior restaurado");
   166→    } catch (e) {
   167→      const err = e as ApiCallError;
   168→      if (err.payload?.type === "nothing_to_undo") toast.info("Nada para desfazer");
   169→      else toast.error(err.payload?.message ?? err.message);
   170→    }
   171→  }, [pushHistory]);
   172→
   173→  // ---------- transform via IA ----------
   174→  const sendRequest = useCallback(
   175→    async (text: string, attachments: Array<{ type: "image"; name: string; data_url: string }>) => {
   176→      if (!projectState) return;
   177→      pushHistory({ role: "user", text });
   178→      setSending(true);
   179→      setCandidate(null);
   180→      try {
   181→        const r: TransformResult = await api.transform({
   182→          request: text,
   183→          base_revision: projectState.revision,
   184→          base_hash: projectState.hash,
   185→          attachments,
   186→        });
   187→        if (r.status === "ready" && r.candidate_project) {
   188→          setCandidate({
   189→            project: r.candidate_project,
   190→            diff: (r.diff ?? { added: [], removed: [], modified: [], counts: { added: 0, removed: 0, modified: 0, total: 0 }, panel_changed: false, info_changed: false }),
   191→            diffSummary: r.diff_summary ?? "",
   192→            baseRevision: r.base_revision,
   193→            baseHash: r.base_hash,
   194→            hash: r.candidate_hash,
   195→          });
   196→          pushHistory({
   197→            role: "assistant",
   198→            text: r.explain,
   199→            meta: {
   200→              status: "ready",
   201→              explain: r.explain,
   202→              assumptions: r.assumptions,
   203→              questions: r.questions,
   204→              diffSummary: r.diff_summary,
   205→              provider: r.meta.provider,
   206→              model: r.meta.model,
   207→              latency: r.meta.latency_ms,
   208→              candidate: r.candidate_project,
   209→            },
   210→          });
   211→        } else {
   212→          pushHistory({
   213→            role: "assistant",
   214→            text: r.explain,
   215→            meta: {
   216→              status: "needs_input",
   217→              explain: r.explain,
   218→              questions: r.questions,
   219→              assumptions: r.assumptions,
   220→              provider: r.meta.provider,
   221→              model: r.meta.model,
   222→              latency: r.meta.latency_ms,
   223→            },
   224→          });
   225→        }
   226→      } catch (e) {
   227→        const err = e as ApiCallError;
   228→        const payload = err.payload;
   229→        pushHistory({
   230→          role: "error",
   231→          text: ERRORS_FRIENDLY[payload?.type ?? ""] ?? payload?.message ?? err.message,
   232→          meta: { errorType: payload?.type ?? "error" },
   233→        });
   234→      } finally {
   235→        setSending(false);
   236→      }
   237→    },
   238→    [projectState, pushHistory],
   239→  );
   240→
   241→  // ---------- preview de candidato (JSON tab / validação) ----------
   242→  const previewCandidate = useCallback(
   243→    async (doc: ProjectDocument): Promise<PreviewResult> => {
   244→      try {
   245→        const r = await api.preview(doc);
   246→        if (r.ok && r.diff && projectState) {
   247→          setCandidate({
   248→            project: doc,
   249→            diff: r.diff,
   250→            diffSummary: r.diff_summary ?? "",
   251→            baseRevision: r.base_revision ?? projectState.revision,
   252→            baseHash: r.base_hash ?? projectState.hash,
   253→            hash: r.candidate_hash ?? "",
   254→          });
   255→          pushHistory({ role: "info", text: `JSON validado — ${r.diff_summary}. Revise o preview 3D e aplique.` });
   256→        }
   257→        return r;
   258→      } catch (e) {
   259→        const err = e as ApiCallError;
   260→        toast.error(err.payload?.message ?? err.message);
   261→        return { ok: false, errors: [{ path: "preview", message: err.message }] };
   262→      }
   263→    },
   264→    [projectState, pushHistory],
   265→  );
   266→
   267→  const applyCandidate = useCallback(async () => {
   268→    if (!candidate) return;
   269→    try {
   270→      // restauração de revisão usa a rota dedicada (source "restore" no log)
   271→      const st = restoreInfo !== null
   272→        ? await api.restoreRevision(restoreInfo)
   273→        : await api.apply({
   274→            candidate: candidate.project,
   275→            base_revision: candidate.baseRevision,
   276→            base_hash: candidate.baseHash,
   277→          });
   278→      setProjectState(st);
   279→      setCandidate(null);
   280→      setRestoreInfo(null);
   281→      setHistoryRefreshKey((k) => k + 1);
   282→      setHistory((h) =>
   283→        h.map((item) =>
   284→          item.role === "assistant" && item.meta?.candidate && !item.meta.applied
   285→            ? { ...item, meta: { ...item.meta, applied: true } }
   286→            : item,
   287→        ),
   288→      );
   289→      pushHistory({
   290→        role: "info",
   291→        text: restoreInfo !== null
   292→          ? `Revisão ${restoreInfo} restaurada como revision ${st.revision}.`
   293→          : `Candidato aplicado — revision ${st.revision}.`,
   294→      });
   295→      toast.success(restoreInfo !== null ? `Revisão ${restoreInfo} restaurada — rev ${st.revision}` : `Aplicado — revision ${st.revision}`);
   296→    } catch (e) {
   297→      const err = e as ApiCallError;
   298→      if (err.status === 409) {
   299→        toast.error("O projeto mudou desde o preview — regenere o candidato.");
   300→        setCandidate(null);
   301→        setRestoreInfo(null);
   302→        void refresh();
   303→      } else {
   304→        toast.error(err.payload?.message ?? err.message);
   305→      }
   306→    }
   307→  }, [candidate, restoreInfo, pushHistory, refresh]);
   308→
   309→  // ---------- comparação com revisão do histórico ----------
   310→  const compareRevision = useCallback(async (rev: number) => {
   311→    if (!projectState) return;
   312→    setComparingRev(rev);
   313→    setRestoreInfo(null);
   314→    try {
   315→      const entry = await api.getRevisionDoc(rev);
   316→      const r = await api.preview(entry.project);
   317→      if (r.ok && r.diff && r.base_revision != null && r.base_hash != null) {
   318→        setCandidate({
   319→          project: entry.project,
   320→          diff: r.diff,
   321→          diffSummary: r.diff_summary ?? "",
   322→          baseRevision: r.base_revision,
   323→          baseHash: r.base_hash,
   324→          hash: r.candidate_hash ?? "",
   325→        });
   326→        setRestoreInfo(rev);
   327→        pushHistory({
   328→          role: "info",
   329→          text: `Comparando rev ${rev} com a atual — ${r.diff_summary}. APLICAR restaura a rev ${rev}; CANCELAR descarta.`,
   330→        });
   331→      } else if (!r.ok) {
   332→        toast.error("Documento da revisão falhou na validação atual.");
   333→      }
   334→    } catch (e) {
   335→      toast.error((e as ApiCallError).payload?.message ?? (e as ApiCallError).message);
   336→    } finally {
   337→      setComparingRev(null);
   338→    }
   339→  }, [projectState, pushHistory]);
   340→
   341→  const cancelCandidate = useCallback(() => {
   342→    setCandidate(null);
   343→    setRestoreInfo(null);
   344→    pushHistory({ role: "info", text: "Preview cancelado — documento atual preservado." });
   345→  }, [pushHistory]);
   346→
   347→  // ---------- status maps para o viewer ----------
   348→  const { statuses, candidateStatuses } = useMemo((): { statuses: StatusMap; candidateStatuses: StatusMap } => {
   349→    if (!projectState) return { statuses: {}, candidateStatuses: {} };
   350→    if (!candidate) return { statuses: {}, candidateStatuses: {} };
   351→    const oldIds = new Set(projectState.project.elements.map((e) => e.id));
   352→    const newIds = new Set(candidate.project.elements.map((e) => e.id));
   353→    const modified = new Set(candidate.diff.modified);
   354→    const added = new Set(candidate.diff.added);
   355→    const removed = new Set(candidate.diff.removed);
   356→    const sOld: StatusMap = {};
   357→    for (const id of oldIds) {
   358→      sOld[id] = removed.has(id) ? "removed" : modified.has(id) ? "modified" : "unchanged";
   359→    }
   360→    const sNew: StatusMap = {};
   361→    for (const id of newIds) {
   362→      sNew[id] = added.has(id) ? "added" : modified.has(id) ? "modified" : "unchanged";
   363→    }
   364→    return { statuses: sOld, candidateStatuses: sNew };
   365→  }, [projectState, candidate]);
   366→
   367→  const selectedElement = useMemo(() => {
   368→    const doc = candidate?.project ?? projectState?.project;
   369→    return doc?.elements.find((e) => e.id === selectedId) ?? null;
   370→  }, [projectState, candidate, selectedId]);
   371→
   372→  const actions = {
   373→    newProject: async () => {
   374→      try {
   375→        const st = await api.newProject();
   376→        setProjectState(st);
   377→        setCandidate(null);
   378→        setRestoreInfo(null);
   379→        setHistoryRefreshKey((k) => k + 1);
   380→        pushHistory({ role: "info", text: "Novo projeto em branco criado." });
   381→      } catch (e) {
   382→        toast.error((e as ApiCallError).message);
   383→      }
   384→    },
   385→    importJson: async (file: File) => {
   386→      try {
   387→        const text = await file.text();
   388→        const doc = JSON.parse(text);
   389→        const st = await api.importProject(doc);
   390→        setProjectState(st);
   391→        setCandidate(null);
   392→        setRestoreInfo(null);
   393→        setHistoryRefreshKey((k) => k + 1);
   394→        pushHistory({ role: "info", text: "JSON importado e validado." });
   395→        toast.success("Projeto importado");
   396→      } catch (e) {
   397→        toast.error((e as ApiCallError).payload?.message ?? (e as Error).message);
   398→      }
   399→    },
   400→    exportPdf: async () => {
   401→      setPdfBusy(true);
   402→      try {
   403→        await api.downloadPdf();
   404→        toast.success("PDF gerado");
   405→      } catch (e) {
   406→        toast.error((e as ApiCallError).payload?.message ?? (e as Error).message);
   407→      } finally {
   408→        setPdfBusy(false);
   409→      }
   410→    },
   411→  };
   412→
   413→  // persiste histórico (sem candidatos antigos pesados)
   414→  useEffect(() => {
   415→    try {
   416→      const slim = history.slice(-30).map((h) => ({
   417→        ...h,
   418→        meta: h.meta?.candidate ? { ...h.meta, candidate: null } : h.meta,
   419→      }));
   420→      localStorage.setItem(HISTORY_KEY, JSON.stringify(slim));
   421→    } catch {
   422→      // quota — ignora
   423→    }
   424→  }, [history]);
   425→
   426→  const toggleHidden = useCallback((id: string) => {
   427→    setHiddenIds((s) => {
   428→      const n = new Set(s);
   429→      if (n.has(id)) n.delete(id);
   430→      else n.add(id);
   431→      return n;
   432→    });
   433→  }, []);
   434→
   435→  const isolateGroup = useCallback((group: string | null) => {
   436→    setIsolatedGroup(group);
   437→    if (!projectState) return;
   438→    const doc = candidate?.project ?? projectState.project;
   439→    if (group === null) {
   440→      setHiddenIds(new Set());
   441→    } else {
   442→      setHiddenIds(new Set(doc.elements.filter((e) => e.group !== group).map((e) => e.id)));
   443→    }
   444→  }, [projectState, candidate]);
   445→
   446→  const focusElement = useCallback((id: string) => {
   447→    setSelectedId(id);
   448→    setFocusRequest({ id, nonce: Date.now() });
   449→  }, []);
   450→
   451→  const fitKey = `${projectState?.revision ?? 0}:${projectState?.hash?.slice(0, 8) ?? ""}:${candidate?.hash.slice(0, 8) ?? "none"}`;
   452→  const project = projectState?.project;
   453→
   454→  return (
   455→    <TooltipProvider>
   456→      <div className="h-screen flex flex-col bg-slate-100 text-slate-900">
   457→        {/* ---------- TOPBAR ---------- */}
   458→        <header className="flex items-center gap-2 px-3 sm:px-4 h-14 bg-[#1b2836] text-slate-100 shrink-0 border-b-4 border-orange-600" role="banner">
   459→          <div className="flex items-center gap-2.5 min-w-0">
   460→            <div className="h-8 w-8 rounded bg-orange-600 grid place-items-center font-black text-white text-sm shrink-0" aria-hidden>
   461→              LC
   462→            </div>
   463→            <div className="leading-tight min-w-0">
   464→              <div className="font-bold text-sm tracking-wide truncate">LED JSON CAD</div>
   465→              <div className="text-[10px] text-slate-400 truncate">documento JSON · mm · Z solo</div>
   466→            </div>
   467→          </div>
   468→
   469→          {project && (
   470→            <div className="hidden md:flex items-center gap-2 ml-4 min-w-0">
   471→              <span className="text-sm font-medium truncate max-w-56" title={project.project.name}>
   472→                {project.project.name}
   473→              </span>
   474→              <Badge variant="outline" className="text-slate-300 border-slate-500">
   475→                rev {projectState!.revision}
   476→              </Badge>
   477→              <Badge variant="outline" className="text-slate-400 border-slate-600 font-mono text-[10px]">
   478→                {projectState!.hash.slice(0, 8)}
   479→              </Badge>
   480→              <Badge variant="outline" className="text-slate-400 border-slate-600">
   481→                {project.elements.length} el.
   482→              </Badge>
   483→            </div>
   484→          )}
   485→
   486→          <div className="ml-auto flex items-center gap-1.5">
   487→            <input
   488→              ref={importFileRef}
   489→              type="file"
   490→              accept=".json,application/json"
   491→              hidden
   492→              onChange={(e) => {
   493→                const f = e.target.files?.[0];
   494→                if (f) void actions.importJson(f);
   495→                e.target.value = "";
   496→              }}
   497→            />
   498→            <TopBtn icon={<FilePlus2 className="h-4 w-4" />} label="NOVO" onClick={() => void actions.newProject()} />
   499→            <TopBtn icon={<Upload className="h-4 w-4" />} label="IMPORTAR" onClick={() => importFileRef.current?.click()} />
   500→            <TopBtn icon={<Undo2 className="h-4 w-4" />} label="DESFAZER" onClick={() => void doUndo()} title="Ctrl+Z" />
   501→            <TopBtn icon={<FileSpreadsheet className="h-4 w-4" />} label="BOM" onClick={() => setBomOpen(true)} />
   502→            <TopBtn
