   104→    };

   106→      if (!dockDragRef.current) return;
   107→      dockDragRef.current = null;
   108→      document.body.style.cursor = "";
   109→      document.body.style.userSelect = "";
   110→      try { localStorage.setItem(DOCK_WIDTH_KEY, String(dockWidthRef.current)); } catch { /* ignore */ }
   111→    };

   113→    window.addEventListener("mouseup", onUp);

   115→      window.removeEventListener("mousemove", onMove);
   116→      window.removeEventListener("mouseup", onUp);

   257→          });

   260→            role: "assistant",
   261→            text: r.explain,
   262→            meta: {
   263→              status: "needs_input",

   265→              questions: r.questions,

   279→          role: "error",
   280→          text: ERRORS_FRIENDLY[payload?.type ?? ""] ?? payload?.message ?? err.message,

   284→        setSending(false);

   480→  const toggleHidden = useCallback((id: string) => {

   500→  const focusElement = useCallback((id: string) => {

   504→
